89inline ArrayDetails validate(const H5::Group& handle, const ritsuko::Version& version, Options& options) {
90 auto dtype = load_scalar_string_attribute(handle, "delayed_type");
91 ArrayDetails output;
92
93 if (dtype == "array") {
94 auto atype = load_scalar_string_attribute(handle, "delayed_array");
95
96 const auto& custom = options.array_validate_registry;
97 auto cit = custom.find(atype);
98 if (cit != custom.end()) {
99 try {
100 output = (cit->second)(handle, version, options);
101 } catch (std::exception& e) {
102 throw std::runtime_error("failed to validate delayed array of type '" + atype + "'; " + std::string(e.what()));
103 }
104
105 } else {
106 static const auto global = default_array_registry();
107 auto git = global.find(atype);
108 if (git != global.end()) {
109 try {
110 output = (git->second)(handle, version, options);
111 } catch (std::exception& e) {
112 throw std::runtime_error("failed to validate delayed array of type '" + atype + "'; " + std::string(e.what()));
113 }
114 } else if (atype.rfind("custom ", 0) != std::string::npos) {
115 try {
116 output = validate_custom_array(handle, version, options);
117 } catch (std::exception& e) {
118 throw std::runtime_error("failed to validate delayed array of type '" + atype + "'; " + std::string(e.what()));
119 }
120 } else if (atype.rfind("external hdf5 ", 0) != std::string::npos && version.lt(1, 1, 0)) {
121 try {
122 output = validate_external_hdf5(handle, version, options);
123 } catch (std::exception& e) {
124 throw std::runtime_error("failed to validate delayed array of type '" + atype + "'; " + std::string(e.what()));
125 }
126 } else {
127 throw std::runtime_error("unknown array type '" + atype + "'");
128 }
129 }
130
131 } else if (dtype == "operation") {
132 auto otype = load_scalar_string_attribute(handle, "delayed_operation");
133
134 const auto& custom = options.operation_validate_registry;
135 auto cit = custom.find(otype);
136 if (cit != custom.end()) {
137 try {
138 output = (cit->second)(handle, version, options);
139 } catch (std::exception& e) {
140 throw std::runtime_error("failed to validate delayed operation of type '" + otype + "'; " + std::string(e.what()));
141 }
142
143 } else {
144 static const auto global = default_operation_registry();
145 auto git = global.find(otype);
146 if (git != global.end()) {
147 try {
148 output = (git->second)(handle, version, options);
149 } catch (std::exception& e) {
150 throw std::runtime_error("failed to validate delayed operation of type '" + otype + "'; " + std::string(e.what()));

173inline ritsuko::Version extract_version(const H5::Group& handle) {
174 ritsuko::Version version;
175
176 if (handle.attrExists("delayed_version")) {
177 auto vstring = load_scalar_string_attribute(handle, "delayed_version");
178 if (vstring == "1.0.0") {
179 version.major = 1;
180 } else {
181 version = ritsuko::parse_version_string(vstring.c_str(), vstring.size(), /* skip_patch = */ true);
182 }
183 } else {
184 version.minor = 99;
185 }
186
187 return version;
188}
