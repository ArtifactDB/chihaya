99 if (cit != custom.end()) {
100 try {
101 output = (cit->second)(group, version, options);

104 }

174inline ritsuko::Version extract_version(const H5::Group& group) {
175 ritsuko::Version version;
176
177 if (group.attrExists("delayed_version")) {
178 auto vstring = load_scalar_string_attribute(group, "delayed_version");
179 if (vstring == "1.0.0") {
180 version.major = 1;
181 } else {
182 version = ritsuko::parse_version_string(vstring.c_str(), vstring.size(), /* skip_patch = */ true);
183 }
184 } else {
185 version.minor = 99;
186 }
187
188 return version;
189}
